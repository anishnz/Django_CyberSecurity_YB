Connect from React

Every answer has the same shape: {"ok": true, "data": …, "ms": 12} or {"ok": false, "error": "…"}. A React dev server may call the public endpoints from these addresses (setting API_CORS_ORIGINS in .env): {% for o in cors %}{{ o }}{% if not forloop.last %}, {% endif %}{% empty %}none{% endfor %}.

// src/api.js – small helper for a React app
const BASE = "{{ base_url }}";

export async function getApi(path) {
  const res = await fetch(BASE + path);
  const body = await res.json();
  if (!body.ok) throw new Error(body.error);
  return body.data;
}
